// cppcheck suppression list for Samd21::I2cDriver static analysis.
//
// Run from this directory with:
//   cppcheck --enable=warning,style,performance,portability --std=c++14 \
//     --language=c++ --inline-suppr --suppressions-list=cppcheck-suppressions.txt \
//     I2cDriver.cpp I2cDriver.hpp I2cDriverHardware.cpp I2cDriverHardware.hpp \
//     I2cDriverHardwareStub.cpp test/ut/*.cpp test/ut/*.hpp
//
// With these suppressions the component analyzes with zero findings.

// Headers for the fprime framework, autocoded base classes and the CMSIS/SAMD21
// device headers are not on cppcheck's include path in a standalone run.
missingInclude
missingIncludeSystem

// Every F Prime component has a single-argument (const char* name) constructor
// generated by the autocoder pattern; a non-explicit ctor here is idiomatic and
// intentional (components are constructed by name).
noExplicitConstructor

// False positive: cppcheck's unusedStructMember cannot see that the HAL status
// structs and the test-only StubState are consumed across other translation
// units (the component .cpp reads the status fields; the unit tests read/write
// StubState). All members are used.
unusedStructMember

// Suppress the "unused suppression" bookkeeping so the list stays portable
// across cppcheck versions / file subsets.
unmatchedSuppression
